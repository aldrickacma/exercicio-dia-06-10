
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.println("Digite N:");
        int n = sc.nextInt();

        double soma = 0;

        for (int i = 1; i <= n; i++) {
            soma += 1.0 / i;

            System.out.print("1/" + i);
            if (i < n) {
                System.out.print(" + ");
            }
        }

        System.out.println();
        System.out.printf("S = %.2f%n", soma);

        sc.close();
    }
}
