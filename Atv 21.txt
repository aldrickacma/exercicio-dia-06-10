
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double minimo = 1621.00;

        for (int i = 1; i <= 10; i++) {
            System.out.println("Codigo do funcionario:");
            int codigo = sc.nextInt();

            System.out.println("Horas trabalhadas:");
            double horas = sc.nextDouble();

            char turno;
            do {
                System.out.println("Turno (M, V ou N):");
                turno = Character.toUpperCase(sc.next().charAt(0));
            } while (turno != 'M' && turno != 'V' && turno != 'N');

            char categoria;
            do {
                System.out.println("Categoria (O ou G):");
                categoria = Character.toUpperCase(sc.next().charAt(0));
            } while (categoria != 'O' && categoria != 'G');

            double percentual;

            if (categoria == 'G' && turno == 'N') {
                percentual = 0.18;
            } else if (categoria == 'G') {
                percentual = 0.15;
            } else if (turno == 'N') {
                percentual = 0.13;
            } else {
                percentual = 0.10;
            }

            double valorHora = minimo * percentual;
            double inicial = horas * valorHora;
            double auxilio;

            if (inicial <= 1080) {
                auxilio = inicial * 0.20;
            } else if (inicial <= 2160) {
                auxilio = inicial * 0.15;
            } else {
                auxilio = inicial * 0.05;
            }

            double finalSalario = inicial + auxilio;

            System.out.println("Codigo: " + codigo);
            System.out.println("Horas: " + horas);
            System.out.println("Valor da hora: R$ " + valorHora);
            System.out.println("Salario inicial: R$ " + inicial);
            System.out.println("Auxilio: R$ " + auxilio);
            System.out.println("Salario final: R$ " + finalSalario);
        }

        sc.close();
    }
}
