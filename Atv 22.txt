
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.println("Quantidade de termos:");
        int n = sc.nextInt();

        long a = 0;
        long b = 1;

        for (int i = 1; i <= n; i++) {
            System.out.print(a + " ");

            long proximo = a + b;
            a = b;
            b = proximo;
        }

        System.out.println();
        sc.close();
    }
}
