
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double somaAdicionais = 0;
        double totalImpostos = 0;
        double maiorPreco = 0;
        double menorPreco = 0;

        int baratos = 0, normais = 0, caros = 0;

        for (int i = 1; i <= 12; i++) {
            System.out.println("Preco unitario:");
            double preco = sc.nextDouble();

            System.out.println("Necessita refrigeracao? (S/N):");
            char refrigeracao =
                    Character.toUpperCase(sc.next().charAt(0));

            System.out.println("Categoria (A/L/V):");
            char categoria =
                    Character.toUpperCase(sc.next().charAt(0));

            double custo;

            if (preco <= 20) {
                if (categoria == 'A') {
                    custo = 2;
                } else if (categoria == 'L') {
                    custo = 3;
                } else {
                    custo = 4;
                }
            } else if (preco <= 50) {
                if (refrigeracao == 'S') {
                    custo = 6;
                } else {
                    custo = 0;
                }
            } else {
                if (refrigeracao == 'S') {
                    if (categoria == 'A') {
                        custo = 5;
                    } else if (categoria == 'L') {
                        custo = 2;
                    } else {
                        custo = 4;
                    }
                } else {
                    if (categoria == 'L') {
                        custo = 1;
                    } else {
                        custo = 0;
                    }
                }
            }

            double imposto;

            if (categoria == 'A' && refrigeracao == 'S') {
                imposto = preco * 0.04;
            } else {
                imposto = preco * 0.02;
            }

            double precoFinal = preco + custo + imposto;

            String classificacao;

            if (precoFinal <= 20) {
                classificacao = "Barato";
                baratos++;
            } else if (precoFinal <= 100) {
                classificacao = "Normal";
                normais++;
            } else {
                classificacao = "Caro";
                caros++;
            }

            System.out.println("Preco unitario: R$ " + preco);
            System.out.println("Custo de estocagem: R$ " + custo);
            System.out.println("Imposto: R$ " + imposto);
            System.out.println("Preco final: R$ " + precoFinal);
            System.out.println("Classificacao: " + classificacao);

            somaAdicionais += custo + imposto;
            totalImpostos += imposto;

            if (i == 1 || precoFinal > maiorPreco) {
                maiorPreco = precoFinal;
            }

            if (i == 1 || precoFinal < menorPreco) {
                menorPreco = precoFinal;
            }
        }

        System.out.println("Media dos adicionais: R$ "
                + (somaAdicionais / 12));

        System.out.println("Maior preco final: R$ " + maiorPreco);
        System.out.println("Menor preco final: R$ " + menorPreco);
        System.out.println("Total de impostos: R$ " + totalImpostos);

        System.out.println("Produtos baratos: " + baratos);
        System.out.println("Produtos normais: " + normais);
        System.out.println("Produtos caros: " + caros);

        sc.close();
    }
}
