public class Main {
    public static void main(String[] args) {

        for (int n = 1; n <= 10; n++) {

            int fatorial = 1;

            for (int i = 1; i <= n; i++) {
                fatorial = fatorial * i;
            }

            System.out.println(n + "! = " + fatorial);
        }
    }
}
